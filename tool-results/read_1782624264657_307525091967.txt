   137→function trackFBQ(event: string, params?: Record<string, unknown>) {
   138→  if (typeof window !== 'undefined' && typeof (window as unknown as Record<string, unknown>).fbq === 'function') {
   139→    (window as unknown as { fbq: (e: string, n: string, p?: Record<string, unknown>) => void }).fbq('trackCustom', event, params);
   140→  }
   141→}

   143→function trackFBStandard(event: string, params?: Record<string, unknown>) {

   145→    (window as unknown as { fbq: (e: string, n: string, p?: Record<string, unknown>) => void }).fbq('track', event, params);
   146→  }

   213→      aria-expanded={open}

   215→    >
   216→      <div className="stitch-faq-question">
